  // stuck in one family look like an idle pool. Neither is reported.

    for(let i=series.length-1;i>0;i--)if(series[i-1].n-series[i].n>=5&&series[i].n<.75*series[i-1].n){series=series.slice(i);break}

    const steps=series.length-1,deltas=series.slice(1).map((value,i)=>value.n-series[i].n),rises=deltas.filter(delta=>delta>0).length,recent=deltas.slice(-3).filter(delta=>delta>0).length,falls=deltas.filter(delta=>delta<0).length;
    const first=series[0].n,last=series.at(-1).n,total=last-first,middle=Math.floor(steps/2),early=series[middle].n-first,late=last-series[middle].n;

  // pair until the map fits: the one with the least mapped and unmapped
  // bytes, so big gaps stay. Heap, stack, program and kernel stay apart.

function memorySlope(points,key){const values=points.filter(point=>isNum(point[key]));if(values.length<2)return null;

function rssTrend(){if(!live.resources?.available||live.resources.stale)return null;
  const all=replayAt===null?resourcePoints():replayResources;if(!all.length)return null;
  const newest=all.at(-1);if(!isNum(newest.rss_bytes)||viewNow()-newest.t>2*RSS_TREND_WINDOWS.at(-1))return null;
  if(!isNum(live.health?.pid)||newest.pid!==live.health.pid)return null;
  let first=all.length-1;while(first>0&&all[first-1].pid===newest.pid)first--;
  const run=all.slice(first).filter(point=>isNum(point.rss_bytes));let best=null;
  for(const window of RSS_TREND_WINDOWS){const points=run.filter(point=>point.t>=newest.t-window);
    if(points.length<RSS_TREND_MIN_POINTS||newest.t-points[0].t<RSS_TREND_MIN_SPAN)continue;
    let rises=0;for(let index=1;index<points.length;index++)if(points[index].rss_bytes>points[index-1].rss_bytes)rises++;
    const steps=points.length-1,slope=memorySlope(points,'rss_bytes'),span=newest.t-points[0].t;
    const steady=rises>=.8*steps&&slope>=2**20&&slope*span/60>=2**20;
    const trend={slope,span,rises,steps,steady};
    if(!best||(steady&&(!best.steady||slope>best.slope))||(!steady&&!best.steady))best=trend}
  return best}

const signalLabel=ended=>ended.signal_name||'signal '+ended.signal;
function exitReason(ended){
  if(ended.reason==='crashed')return [`Crashed: ${signalLabel(ended)}`,'critical',CRASH_SIGNALS[ended.signal_name]||'A signal that reports a fault in the process ended it.'];
  if(ended.reason==='killed')return [`Killed by ${signalLabel(ended)}`,'warning',(KILL_SIGNALS[ended.signal_name]||'Another process sent a signal that ended it.')+(ended.signal_name==='SIGKILL'&&ended.kernel_log!=='read'?' Without the kernel log, an out-of-memory kill looks the same.':'')];
  if(ended.reason==='exited')return ended.exit_code===0?['Exited normally','good','The process returned from main or called exit(0).']:[`Exited with code ${ended.exit_code}`,'warning','The process called exit() or returned from main with a non-zero code, usually after it logged an error.'];
  return EXIT_REASONS[ended.reason]||[ended.reason,'warning',''];
}
function exitStatusText(ended){
  if(isNum(ended.signal))return `Exit status: ${signalLabel(ended)} (signal ${ended.signal})${ended.core_dumped?', core dumped':''}.`;
  if(isNum(ended.exit_code))return `Exit status: exit code ${ended.exit_code}.`;
  return 'Exit status: not available. The sampler reads it on Linux 6.15 and later, or while the process is a zombie.'}
